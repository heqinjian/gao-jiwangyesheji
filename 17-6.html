// 流程日志：

核心异步代码结构

function hqj_wait(秒, 名称) {
  return new Promise(function (resolve) {
    setTimeout(function () { resolve(名称); }, 秒 * 1000);
  });
}

async function hqj_main() {
  await hqj_wait(6, "买原料");   // 串行：完成才往下走
  await hqj_wait(1, "取原料");
  await Promise.all([              // 并行：同时开始
    hqj_wait(4, "和面"),
    hqj_wait(4, "调馅")
  ]);
  await hqj_wait(3, "包饺子");
  await hqj_wait(2, "吃饺子");
}
